import React, { useState } from 'react';
import { SafeAreaView, ScrollView, StatusBar, StyleSheet, Text, TouchableOpacity, View } from 'react-native';
import { Ionicons } from '@expo/vector-icons';

type Tab = 'Heute' | 'Einsätze' | 'Zeit' | 'Aufgaben' | 'Profil';
type Icon = React.ComponentProps<typeof Ionicons>['name'];
const NAV: { label: Tab; icon: Icon }[] = [
  { label: 'Heute', icon: 'home-outline' }, { label: 'Einsätze', icon: 'calendar-outline' },
  { label: 'Zeit', icon: 'timer-outline' }, { label: 'Aufgaben', icon: 'checkbox-outline' },
  { label: 'Profil', icon: 'person-outline' },
];
const palette = { navy: '#12243B', muted: '#778599', blue: '#315CF5', bg: '#F4F7FB', border: '#E5EBF2' };
function Tile({ icon, title, detail, onPress }: { icon: Icon; title: string; detail: string; onPress?: () => void }) {
  return <TouchableOpacity onPress={onPress} activeOpacity={0.8} style={styles.tile}>
    <View style={styles.tileIcon}><Ionicons name={icon} size={21} color={palette.blue} /></View>
    <View style={{ flex: 1 }}><Text style={styles.tileTitle}>{title}</Text><Text style={styles.muted}>{detail}</Text></View>
    <Ionicons name="chevron-forward" size={17} color="#A3ADBA" />
  </TouchableOpacity>;
}
export default function App() {
  const [tab, setTab] = useState<Tab>('Heute');
  const [running, setRunning] = useState(false);
  const [completed, setCompleted] = useState<string[]>([]);
  const [message, setMessage] = useState('');
  const tasks = ['Arbeitsmaterial prüfen', 'Einsatzbestätigung vorbereiten', 'Dokumente aktualisieren'];
  const toggleTask = (task: string) => setCompleted(prev => prev.includes(task) ? prev.filter(t => t !== task) : [...prev, task]);
  const showNotice = (text: string) => setMessage(text);
  return <SafeAreaView style={styles.root}>
    <StatusBar barStyle="dark-content" />
    <View style={styles.header}><View><Text style={styles.brand}>HSU <Text style={{ color: palette.blue }}>•</Text> MITARBEITER</Text><Text style={styles.subtitle}>Dein Arbeitsbereich</Text></View><TouchableOpacity style={styles.avatar} onPress={() => setTab('Profil')}><Ionicons name="person" size={20} color={palette.blue} /></TouchableOpacity></View>
    <ScrollView contentContainerStyle={styles.content}>
      {tab === 'Heute' && <>
        <Text style={styles.eyebrow}>DEIN ÜBERBLICK</Text><Text style={styles.heading}>Guten Tag 👋</Text><Text style={styles.intro}>Alles Wichtige für deinen Arbeitstag.</Text>
        <View style={styles.hero}><Text style={styles.heroLabel}>NÄCHSTER EINSATZ</Text><Text style={styles.heroTitle}>Noch kein Einsatz zugewiesen</Text><Text style={styles.heroSub}>Neue Einsätze erscheinen hier, sobald die Verbindung eingerichtet ist.</Text><TouchableOpacity style={styles.heroButton} onPress={() => setTab('Einsätze')}><Text style={styles.heroButtonText}>Einsätze ansehen →</Text></TouchableOpacity></View>
        <View style={styles.metrics}><View style={styles.metric}><Text style={styles.metricNumber}>—</Text><Text style={styles.muted}>Wochenstunden</Text></View><View style={styles.metric}><Text style={styles.metricNumber}>{completed.length}/{tasks.length}</Text><Text style={styles.muted}>Demo-Aufgaben</Text></View></View>
        <Text style={styles.section}>Schnellzugriff</Text><Tile icon="timer-outline" title="Zeiterfassung" detail={running ? 'Demo-Timer läuft' : 'Arbeitszeit erfassen'} onPress={() => setTab('Zeit')} /><Tile icon="checkbox-outline" title="Meine Aufgaben" detail="Checkliste öffnen" onPress={() => setTab('Aufgaben')} /><Tile icon="document-text-outline" title="Dokumente" detail="Noch nicht verbunden" onPress={() => showNotice('Dokumente benötigen eine sichere Backend-Verbindung.')} /><Tile icon="chatbubbles-outline" title="Nachrichten" detail="Noch nicht verbunden" onPress={() => showNotice('Nachrichten benötigen eine sichere Backend-Verbindung.')} />
      </>}
      {tab === 'Einsätze' && <><Text style={styles.heading}>Meine Einsätze</Text><Text style={styles.intro}>Deine Schichten und Termine an einem Ort.</Text><View style={styles.empty}><Ionicons name="calendar-outline" size={36} color={palette.blue} /><Text style={styles.emptyTitle}>Noch keine Einsätze</Text><Text style={styles.muted}>Einsatzdaten werden nach Anbindung des Backends geladen.</Text></View></>}
      {tab === 'Zeit' && <><Text style={styles.heading}>Zeiterfassung</Text><Text style={styles.intro}>Deine Arbeitszeit im Blick.</Text><View style={styles.timeCard}><Ionicons name="time-outline" size={42} color={palette.blue} /><Text style={styles.timeTitle}>{running ? 'Demo-Erfassung aktiv' : 'Bereit zum Starten'}</Text><Text style={styles.muted}>Testmodus: Zeiten werden nicht gespeichert oder übertragen.</Text><TouchableOpacity style={styles.primary} onPress={() => setRunning(!running)}><Text style={styles.primaryText}>{running ? 'Demo stoppen' : 'Demo starten'}</Text></TouchableOpacity></View></>}
      {tab === 'Aufgaben' && <><Text style={styles.heading}>Meine Aufgaben</Text><Text style={styles.intro}>Interaktive Beispielaufgaben – lokal auf diesem Bildschirm.</Text>{tasks.map(task => <TouchableOpacity key={task} style={styles.tile} onPress={() => toggleTask(task)}><Ionicons name={completed.includes(task) ? 'checkmark-circle' : 'ellipse-outline'} color={palette.blue} size={24} /><Text style={[styles.tileTitle, { flex: 1, marginLeft: 12 }, completed.includes(task) && { textDecorationLine: 'line-through' }]}>{task}</Text></TouchableOpacity>)}</>}
      {tab === 'Profil' && <><Text style={styles.heading}>Mein Profil</Text><Text style={styles.intro}>Persönliche Einstellungen und Informationen.</Text><Tile icon="person-circle-outline" title="Mitarbeiterprofil" detail="Anmeldung noch nicht eingerichtet" /><Tile icon="shield-checkmark-outline" title="Datenschutz & Sicherheit" detail="Rollen und Berechtigungen folgen" /><Tile icon="wallet-outline" title="Lohn & Abrechnungen" detail="Noch nicht verbunden" /><Tile icon="help-circle-outline" title="Hilfe & Support" detail="Kontakt wird eingerichtet" /></>}
      {!!message && <TouchableOpacity style={styles.notice} onPress={() => setMessage('')}><Text style={styles.noticeText}>{message}  ✕</Text></TouchableOpacity>}
    </ScrollView>
    <View style={styles.nav}>{NAV.map(item => <TouchableOpacity key={item.label} onPress={() => { setTab(item.label); setMessage(''); }} style={styles.navItem}><Ionicons name={item.icon} size={23} color={tab === item.label ? palette.blue : '#94A0B0'} /><Text style={[styles.navText, tab === item.label && { color: palette.blue, fontWeight: '700' }]}>{item.label}</Text></TouchableOpacity>)}</View>
  </SafeAreaView>;
}

});
